170: function carregarScriptX(): Promise<void> {
171:   if (typeof window === "undefined") return Promise.resolve();
172:   if (window.twttr?.widgets) return Promise.resolve();
173:   if (xScriptPromise) return xScriptPromise;
174: 
175:   xScriptPromise = new Promise<void>((resolve, reject) => {
176:     let script = document.getElementById("x-wjs") as HTMLScriptElement | null;
177: 
178:     const aguardarWidgets = () => {
179:       const inicio = Date.now();
180:       const timer = window.setInterval(() => {
181:         if (window.twttr?.widgets) {
182:           window.clearInterval(timer);
183:           resolve();
184:           return;
185:         }
186: 
187:         if (Date.now() - inicio > 8000) {
188:           window.clearInterval(timer);
189:           reject(new Error("widgets.js do X não ficou disponível."));
190:         }
191:       }, 120);
192:     };
193: 
194:     if (!script) {
195:       script = document.createElement("script");
196:       script.id = "x-wjs";
197:       script.src = "https://platform.x.com/widgets.js";
198:       script.async = true;
199:       script.charset = "utf-8";
200:       document.body.appendChild(script);
201:     }
202: 
203:     if (script.dataset.loaded === "true") {
204:       aguardarWidgets();
205:       return;
206:     }
207: 
208:     script.addEventListener(
209:       "load",
210:       () => {
211:         if (script) script.dataset.loaded = "true";
212:         aguardarWidgets();
213:       },
214:       { once: true }
215:     );
216:     script.addEventListener("error", () => reject(new Error("Falha ao carregar widgets.js do X.")), {
217:       once: true,
218:     });
219:   }).catch((error) => {
220:     xScriptPromise = null;
221:     throw error;
222:   });
223: 
224:   return xScriptPromise!;
